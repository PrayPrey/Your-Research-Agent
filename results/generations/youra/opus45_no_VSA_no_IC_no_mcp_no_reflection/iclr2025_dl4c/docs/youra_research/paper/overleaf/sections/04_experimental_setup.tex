\section{Experimental Setup}

\textbf{Research Questions:}
\begin{itemize}
    \item RQ1: Does HIGH reach pass@1 $>$ 0.3 faster than LOW?
    \item RQ2: Does MEDIUM outperform LOW?
    \item RQ3: Does error-type distribution differ?
\end{itemize}

\textbf{Evaluation:}
\begin{itemize}
    \item Primary: pass@1 on validation subset
    \item Statistical: t-test, Cohen's d, $p < 0.05$
\end{itemize}

\textbf{Missing diagnostics (limitation):}
\begin{itemize}
    \item No training loss curves recorded
    \item No KL divergence or entropy monitoring
    \item No intermediate checkpoints
    \item No zero-shot baseline verification in our evaluation setup
\end{itemize}
