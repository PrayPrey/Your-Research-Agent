\section{Experiments}

\subsection{Research Questions}

\begin{description}
    \item[RQ1 (Feasibility):] Can both metrics be reliably computed?
    \item[RQ2 (Entropy Signal):] Does higher entropy correlate with incorrect answers?
    \item[RQ3 (Consistency Signal):] Does lower consistency correlate with incorrect answers?
    \item[RQ4 (Fusion Benefit):] Does combining signals improve over the best single metric?
\end{description}
