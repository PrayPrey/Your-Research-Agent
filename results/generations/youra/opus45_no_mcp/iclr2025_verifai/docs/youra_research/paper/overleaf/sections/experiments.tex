\subsection{Research Questions}

\textbf{RQ1:} Do static and execution feedback detect overlapping error classes?

\textbf{RQ2:} What proportion of test-failing code contains structural errors?

\textbf{RQ3:} What proportion of static-clean code fails execution tests?

\subsection{Evaluation Metrics}

\begin{itemize}
    \item \textbf{Jaccard Similarity:} Threshold $< 0.3$ for orthogonality
    \item \textbf{Structural Coverage:} $> 60\%$ indicates reliable static detection
    \item \textbf{Behavioral Rate:} $> 40\%$ indicates execution catches missed errors
\end{itemize}

\subsection{Implementation}

\begin{itemize}
    \item Static: pylint 2.17.x, mypy 1.4.x, 30s timeout
    \item Execution: EvalPlus harness, 5s timeout, 8 workers
    \item Compute: Standard CPU, $\sim$10 minutes total
\end{itemize}
