\subsubsection{Evaluation Metrics}

\begin{itemize}
\item \textbf{Generalization Gap:} gap = acc\_in-domain - acc\_held-out
\item \textbf{Paper Ratio:} High-use / low-use optimization paper counts
\item \textbf{Texture Bias Ratio:} texture\_accuracy / (texture\_accuracy + shape\_accuracy)
\end{itemize}

\subsubsection{Statistical Tests}

\begin{itemize}
\item \textbf{H-M1:} Mann-Whitney U test (one-sided) for paper count differences between groups
\item \textbf{H-E1:} Direction check (proof-of-concept); Cohen's d requires multi-run replication
\end{itemize}
