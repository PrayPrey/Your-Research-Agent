\section{Methodology}

\subsection{Trajectory Metrics}

\textbf{Normalized Trajectory Instability (NTI)}:
\begin{equation}
\text{NTI} = \frac{\sigma(\{H_l\}_{l=24}^{31})}{\mu(\{H_l\}_{l=24}^{31}) + \epsilon}
\end{equation}

\textbf{Convergence Monotonicity Index (CMI)}:
\begin{equation}
\text{CMI} = \frac{1}{L-1} \sum_{l=24}^{30} \mathbb{1}[s_{l+1} > s_l - \delta]
\end{equation}

\textbf{Representational Competition Index (RCI)}: Binary indicator of top-token change between consecutive layers.

\subsection{Model and Dataset}

\begin{itemize}
    \item \textbf{Model}: LLaMA-2-7B (meta-llama/Llama-2-7b-hf)
    \item \textbf{Dataset}: TruthfulQA MC1 (817 questions, 4114 prompt-choice pairs)
    \item \textbf{Evaluation}: 5-fold stratified CV, logistic regression, AUROC
\end{itemize}
