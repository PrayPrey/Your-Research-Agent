\section{Experimental Setup}

\subsection{Datasets}

\begin{itemize}
    \item \textbf{HH-RLHF:} 40,688 responses (test split, ``helpful'' subset)
    \item \textbf{RewardBench Safety:} 1,208 responses
    \item \textbf{Total:} 41,896 responses
\end{itemize}

\subsection{Implementation}

\begin{itemize}
    \item \textbf{Proxy Detection:} TF-IDF (n-gram 1--2, max 5000 features) + Logistic Regression
    \item \textbf{Reward Scoring:} OpenAssistant/reward-model-deberta-v3-large-v2
    \item \textbf{Adversarial Probing:} 4096-dimensional simulated hidden states, 3 seeds, 5 epochs
    \item \textbf{Semantic Clustering:} all-MiniLM-L6-v2, UMAP (5 components), BERTopic
\end{itemize}

\subsection{Evaluation Metrics}

\begin{itemize}
    \item \textbf{H-E1:} Proxy AUROC $\geq$ 0.8
    \item \textbf{H-M1:} BAI AUROC $\geq$ 0.7 after GRL, $R^2$ degradation $<$ 2\%
    \item \textbf{H-M2:} Disagreement rate $\geq$ 20\% (PASS) or $\geq$ 10\% (PARTIAL)
    \item \textbf{H-C1:} Agency pattern rate $\geq$ 50\%
\end{itemize}
