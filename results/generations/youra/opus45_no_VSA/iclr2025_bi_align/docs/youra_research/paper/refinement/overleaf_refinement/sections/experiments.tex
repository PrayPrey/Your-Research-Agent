
\subsubsection{4.1 Datasets}

\begin{itemize}
\item \textbf{HH-RLHF:} 40,688 responses (test split, ''helpful'' subset)
\item \textbf{RewardBench Safety:} 1,208 responses
\item \textbf{Total:} 41,896 responses
\end{itemize}

\subsubsection{4.2 Implementation}

\begin{itemize}
\item \textbf{Proxy Detection:} TF-IDF (n-gram 1-2, max 5000 features) + Logistic Regression (C=1.0)
\item \textbf{Train/Test Split:} 33,516 / 8,380 (80/20)
\item \textbf{Reward Scoring:} OpenAssistant/reward-model-deberta-v3-large-v2, batch size 32, max tokens 512
\item \textbf{Adversarial Probing:} 4096-dimensional simulated hidden states (Llama-3-8B equivalent), 3 seeds (42, 123, 456), 5 epochs, 2000 samples
\item \textbf{Semantic Clustering:} all-MiniLM-L6-v2, UMAP (5 components), BERTopic with HDBSCAN (min\_cluster\_size=50)
\end{itemize}
