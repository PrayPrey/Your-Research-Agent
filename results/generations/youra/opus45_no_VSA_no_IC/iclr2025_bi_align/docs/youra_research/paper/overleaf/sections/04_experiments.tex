\section{Experiments}

\subsection{Experiment 1: Mode 3 Existence (H-E1)}

\textbf{Objective}: Determine whether Mode~3 constitutes a substantial proportion of preference battles.

\textbf{Setup}: 57,477 battles, median-split classification on entropy and variance dimensions.

\textbf{Gate}: MUST\_WORK if Mode~3 $>$ 10\%.

\subsection{Experiment 2: Semantic Similarity (H-M1)}

\textbf{Objective}: Test whether Mode~3 arises from semantic divergence between response pairs.

\textbf{Setup}: all-MiniLM-L6-v2 embeddings, cosine similarity computed per response pair.

\textbf{Prediction}: Mode~3 similarity $<$ Mode~1, with Cohen's $d > 0.3$.

\subsection{Experiment 3: Prompt Type (H-C1)}

\textbf{Objective}: Test whether Mode~3 concentrates in subjective versus objective tasks.

\textbf{Setup}: Keyword-based prompt classification into subjective (creative writing, opinions) and objective (coding, math, factual) categories.

\textbf{Prediction}: Subjective/Objective ratio $> 1.5$.
