\section{Results}

\subsection{H-E1: Mode 3 Existence --- CONFIRMED}

\begin{table}[h]
\centering
\caption{Mode distribution across 57,477 Chatbot Arena battles.}
\label{tab:mode_distribution}
\begin{tabular}{@{}lrr@{}}
\toprule
Mode & Count & Proportion \\
\midrule
1 (Aligned-Confident) & 15,107 & 26.3\% \\
2 (Aligned-Uncertain) & 13,714 & 23.9\% \\
\textbf{3 (Misaligned-Confident)} & \textbf{13,632} & \textbf{23.7\%} \\
4 (Misaligned-Uncertain) & 15,024 & 26.1\% \\
\bottomrule
\end{tabular}
\end{table}

Mode~3 = \textbf{23.7\%} (95\% CI: 23.4--24.0\%, $p<0.001$). Far exceeds the 10\% threshold.

\subsection{H-M1: Semantic Similarity --- FALSIFIED}

\begin{table}[h]
