
\subsection{Research Question}

Can EVAF produce filtered AI feedback with accept rate in the viable 20--60\% range?

\subsection{Dataset}

\textbf{HumanEval}: 164 Python programming problems with unit tests \citep{chen2021humaneval}.

\subsection{Models}

\begin{itemize}
    \item \textbf{Baseline}: CodeT5-770M (Salesforce/codet5-large) \citep{wang2021codet5}
    \item \textbf{Feedback}: CodeLlama-7b-Instruct (codellama/CodeLlama-7b-Instruct-hf) \citep{roziere2023codellama}
\end{itemize}

\subsection{Evaluation Protocol}

\begin{enumerate}
    \item Generate baseline code for all 164 problems
    \item Filter to failing problems
    \item Run EVAF gating on each failing problem
    \item Compute accept rate and secondary metrics
\end{enumerate}

\subsection{Success Criteria}

\textbf{Pass}: Accept rate 20--60\%\\
\textbf{Fail}: Accept rate $<$10\% or $>$90\%
