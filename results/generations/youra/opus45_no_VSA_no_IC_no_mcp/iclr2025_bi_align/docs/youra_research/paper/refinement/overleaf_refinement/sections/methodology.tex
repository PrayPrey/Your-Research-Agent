\subsubsection{3.2 Experimental Setup}

\textbf{Model:} Llama-2-7B (meta-llama/Llama-2-7b-hf) with LoRA adaptation (rank=16, alpha=32)

\textbf{Training Data:} Anthropic HH-RLHF dataset (15,000 training samples)

\textbf{Evaluation Benchmarks:}
\begin{itemize}
\item TruthfulQA MC1 (817 samples)
\item HHH-helpful (1,000 samples)
\item HHH-harmless (1,000 samples)
\end{itemize}

\textbf{RLHF Configuration:}
\begin{itemize}
\item Loss: Bradley-Terry with center\_rewards coefficient = 0.01
\item Learning rate: 1e-4
\item Batch size: 4 (effective 16 with gradient accumulation)
\item Epochs: 1
\end{itemize}

\textbf{DPO Configuration:}
\begin{itemize}
\item Beta (sharpness control): 0.1
\item Learning rate: 5e-7
\item Batch size: 2 (effective 16 with gradient accumulation)
\item Epochs: 1
\end{itemize}

\subsubsection{3.3 Metrics}

\textbf{H-M1 (RLHF Smoothness):}
\begin{itemize}
\item Reward output range (continuous distribution expected)
\item Evaluation accuracy (> 50\% indicating learned preferences)
\item Evaluation margin (positive value indicating preference direction)
\end{itemize}

\textbf{H-M2 (DPO Sharpness):}
\begin{itemize}
\item Sharpness ratio = DPO margin standard deviation / RLHF margin standard deviation
\item Boundary accuracy on cases where RLHF margin < 0.1
\item Confident ratio (proportion of boundary cases with |margin| > 0.1)
\end{itemize}

\textbf{H-M3 (Attractors):}
\begin{itemize}
\item Clustering gap = within-method similarity - cross-method similarity
\item Silhouette score for K-means clustering by method
\item Statistical significance via permutation test
\end{itemize}

\textbf{H-M4 (Differential Profiles):}
\begin{itemize}
\item Primary criterion: max|Cohen's d| > 0.3 AND min|Cohen's d| < 0.15
\item Profile correlation between DPO and RLHF benchmark performance vectors
\end{itemize}
