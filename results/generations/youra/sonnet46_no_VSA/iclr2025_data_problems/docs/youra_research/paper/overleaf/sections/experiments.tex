\section{Experimental Setup}
\label{sec:setup}

\subsection{Research Questions}

\textbf{RQ1:} Does global $k$-th percentile thresholding on \texttt{ccnet\_perplexity} produce statistically significant language-group retention disparity (Cram{\'e}r's $V > 0.10$ with Holm $p < 0.05$)?

\textbf{RQ2:} Is the disparity consistent across the full practical range of threshold aggressiveness ($k \in \{10, 20, 30, 40, 50\}$)?

\textbf{RQ3:} Does the retention ordering align with the CCNet KenLM mechanism prediction --- Germanic (en, de) systematically below Romance (es, fr, it)?

\subsection{Evaluation Metrics}

\begin{itemize}
\item \textbf{Cram{\'e}r's $V$} (primary): language $\times$ retention association strength
\item \textbf{Per-language retention rates}: human-interpretable view per language per $k$
\item \textbf{Max--min retention gap}: worst-case disparity between any two language groups
\item \textbf{Statistical significance}: Holm-Bonferroni corrected chi-square $p$-values
\end{itemize}
