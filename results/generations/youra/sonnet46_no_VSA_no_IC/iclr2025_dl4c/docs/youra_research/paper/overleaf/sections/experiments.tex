\section{Experimental Setup}
\label{sec:experiments}

\subsection{Research Questions}

\textbf{RQ1 (Existence):} Does MBPP exhibit a non-degenerate binary execution reward variance distribution for DeepSeek-Coder-7B-Instruct? Is there a meaningful subset with variance $> 0.1$? \textit{(h-e1)}

\textbf{RQ2 (Selection Signal):} Does top-50 variance-guided selection achieve statistically significantly higher mean variance than random-50? \textit{(h-m1)}

\textbf{RQ3 (Gradient Concentration):} Does GRPO training on the variance-selected subset produce lower \texttt{frac\_reward\_zero\_std} than random-50 selection? \textit{(h-m2, h-m3)}

\subsection{Dataset and Model}

\textbf{Dataset:} MBPP training split \citep{Austin2021MBPP} --- 374 Python programming problems (\texttt{google-research-datasets/mbpp}, subset=full, split=train). MBPP is the standard benchmark for binary execution reward RLEF \citep{Gehring2024RLEF}.

\textbf{Model:} DeepSeek-Coder-7B-Instruct \citep{Guo2024DeepSeekCoder} --- a 7B-parameter instruction-tuned code generation model, confirmed functional with TRL GRPOTrainer in our environment.

\subsection{Evaluation Metrics}

\begin{itemize}
\item \texttt{count\_nonzero\_variance} (gate: $\geq$15 problems with $v_i > 0.1$)
\item \texttt{mean\_var\_selected} vs \texttt{mean\_var\_random} (Mann-Whitney U, one-sided, $\alpha=0.05$)
\item \texttt{frac\_reward\_zero\_std} per training step (primary training diagnostic)
\item HumanEval+ pass@1 via EvalPlus \citep{Liu2023EvalPlus} (planned; not conducted --- cold-start prevented evaluation)
\end{itemize}
